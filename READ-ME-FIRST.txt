একজন Guardian-এর একাধিক সন্তান থাকার সমস্যা সমাধান হলো।

=========================================================
ধাপ ১ — Security Rules বদলান (isGuardianOf ফাংশন)
=========================================================
আগেরটার সাথে শুধু isGuardianOf() ফাংশন বদলেছে (এখন array-ও চেক করে,
আগের একক-সন্তান অ্যাকাউন্টও ভাঙবে না) — বাকি সব অপরিবর্তিত। সব মুছে
এটা দিয়ে Publish করুন:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isSuperAdmin() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin' &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.adminLevel == 'super';
    }

    function isTeacher() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'teacher';
    }

    function isGuardianOf(studentUid) {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        (
          get(/databases/$(database)/documents/users/$(request.auth.uid)).data.linkedStudentUid == studentUid ||
          (
            'linkedStudentUids' in get(/databases/$(database)/documents/users/$(request.auth.uid)).data &&
            get(/databases/$(database)/documents/users/$(request.auth.uid)).data.linkedStudentUids.hasAny([studentUid])
          )
        );
    }

    match /admissions/{applicationId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /notices/{noticeId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /users/{userId} {
      allow read: if (request.auth != null && request.auth.uid == userId)
        || isAdmin()
        || (isTeacher() && resource.data.role == 'student')
        || isGuardianOf(userId);
      allow write: if false;
    }

    match /attendance/{attendanceId} {
      allow create, update: if isAdmin() || isTeacher();
      allow read: if isAdmin() || isTeacher()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /assessments/{assessmentId} {
      allow create, update: if isAdmin() || isTeacher();
      allow read: if isAdmin() || isTeacher()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /homework/{homeworkId} {
      allow create, update, delete: if isAdmin() || isTeacher();
      allow read: if request.auth != null;
    }

    match /fees/{feeId} {
      allow create, update, delete: if isSuperAdmin();
      allow read: if isSuperAdmin()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /schedule/{scheduleId} {
      allow create, update, delete: if isAdmin();
      allow read: if request.auth != null;
    }

    match /successStories/{storyId} {
      allow create, update, delete: if isAdmin();
      allow read: if resource.data.published == true || isAdmin();
    }

    match /scholarshipApplications/{applicationId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /blogPosts/{slug} {
      allow read: if resource.data.published == true || isAdmin();
      allow create, update, delete: if isAdmin();
    }

    match /resources/{resourceId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /contactMessages/{messageId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /practicalLearning/{entryId} {
      allow read: if resource.data.published == true || isAdmin();
      allow create, update, delete: if isAdmin();
    }
  }
}

=========================================================
ধাপ ২ — ফাইল বসান (৬টা)
=========================================================
সব Merge/Replace করে বসিয়ে push করুন।

=========================================================
কীভাবে কাজ করে
=========================================================
- নতুন Guardian অ্যাকাউন্ট তৈরির সময় এখন একাধিক শিক্ষার্থী টিক
  দিয়ে বেছে নেওয়া যাবে (আগে শুধু ১ জন)
- "তথ্য এডিট" থেকে কোনো Guardian-এর সন্তানের তালিকা বদলানো (যোগ/বাদ)
  যাবে
- Guardian Dashboard-এ এখন ১-এর বেশি সন্তান থাকলে উপরে একটা "সন্তান
  বেছে নিন" অংশ দেখাবে — যাকে বেছে নেওয়া হবে তার Attendance,
  Assessment, ফি — সব সেই অনুযায়ী দেখাবে
- ১ জন সন্তান হলে আগের মতোই সরাসরি দেখাবে, কোনো বাড়তি ধাপ লাগবে না
- আগে থেকে তৈরি করা Guardian অ্যাকাউন্ট (১ সন্তান ধরে বানানো) ভাঙবে
  না, স্বাভাবিকভাবে কাজ করতে থাকবে

=========================================================
টেস্ট করবেন যেভাবে
=========================================================
1. দুইজন টেস্ট শিক্ষার্থী তৈরি করুন (যদি না থাকে)
2. একটা নতুন Guardian অ্যাকাউন্ট বানান, দুজনকেই টিক দিন
3. সেই Guardian দিয়ে লগইন করে dashboard-এ "সন্তান বেছে নিন" অংশ
   দেখা যাচ্ছে কিনা দেখুন
4. একজন থেকে আরেকজনে ক্লিক করে বদলে দেখুন Attendance/Assessment/ফি
   ঠিকভাবে বদলাচ্ছে কিনা
5. আগে থেকে থাকা কোনো পুরনো Guardian অ্যাকাউন্ট দিয়ে লগইন করে
   নিশ্চিত হোন সেটাও আগের মতোই কাজ করছে
