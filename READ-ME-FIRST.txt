=========================================================
ধাপ ১ — Security Rules-এ আরেকটা collection যোগ করুন
=========================================================
Firestore Database -> Rules -এ গিয়ে সব মুছে এটা দিয়ে বদলে Publish করুন
(আগেরটার সাথে শুধু "assessments" ব্লকটা নতুন যোগ হলো):

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isTeacher() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'teacher';
    }

    function isGuardianOf(studentUid) {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.linkedStudentUid == studentUid;
    }

    match /admissions/{applicationId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /notices/{noticeId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /users/{userId} {
      allow read: if (request.auth != null && request.auth.uid == userId)
        || isAdmin()
        || (isTeacher() && resource.data.role == 'student')
        || isGuardianOf(userId);
      allow write: if false;
    }

    match /attendance/{attendanceId} {
      allow create, update: if isAdmin() || isTeacher();
      allow read: if isAdmin() || isTeacher()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /assessments/{assessmentId} {
      allow create, update: if isAdmin() || isTeacher();
      allow read: if isAdmin() || isTeacher()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }
  }
}

১) "Teacher অ্যাকাউন্ট তৈরি করতে গিয়ে 'কিছু একটা সমস্যা হয়েছে' লেখা
   error" — এই আপডেটের পর একই সমস্যা হলে এবার message-টা আরও স্পষ্ট
   হবে এবং ঠিক কী করতে হবে বলে দেবে। সবচেয়ে সম্ভাব্য কারণ:
   - Vercel-এ FIREBASE_SERVICE_ACCOUNT_KEY অথবা ADMIN_EMAILS যোগ করা
     হয়নি বা ভুল ফরম্যাটে আছে (আগের আপডেটের ধাপ ১-২ দেখুন)
   - এই দুটো env variable যোগ/পরিবর্তন করার পর Vercel-এ নতুন করে
     Deploy করতে হবে (Redeploy না করলে পুরনো মান দিয়েই চলে) —
     Deployments ট্যাবে গিয়ে সর্বশেষ deployment-এর ৩-ডট মেনুতে
     "Redeploy" চাপুন
   এরপরও সমস্যা হলে Vercel -> আপনার প্রজেক্ট -> Logs ট্যাবে গিয়ে
   /api/admin/create-user -এ ক্লিক করলে যে আসল error message দেখাবে,
   সেটার স্ক্রিনশট আমাকে দিন — এবার আমি লগে বিস্তারিত লিখে রাখছি,
   তাই কারণ ঠিক বের করা যাবে

২) "/teacher/... এ 404" — এটা কোডের বাগ না, বরং মনে হচ্ছে আগের কোনো
   একটা আপডেট (৭, ১১, বা ১২ নম্বর) পুরোপুরি push হয়নি। GitHub-এ গিয়ে
   আপনার রিপোর src/app/teacher ফোল্ডারে login, dashboard, attendance,
   এই তিনটা ফোল্ডার আছে কিনা দেখুন — না থাকলে বুঝবেন কোনো একটা আপডেট
   বাদ পড়ে গেছে, আগেরগুলো আরেকবার বসিয়ে push করতে হবে
