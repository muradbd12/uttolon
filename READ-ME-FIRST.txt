=========================================================
ধাপ ১ — Security Rules-এ আরেকটা collection যোগ করুন
=========================================================
আগেরটার সাথে শুধু "successStories" ব্লক নতুন — সব মুছে এটা দিয়ে
Publish করুন:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isTeacher() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'teacher';
    }

    function isGuardianOf(studentUid) {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.linkedStudentUid == studentUid;
    }

    match /admissions/{applicationId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /notices/{noticeId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /users/{userId} {
      allow read: if (request.auth != null && request.auth.uid == userId)
        || isAdmin()
        || (isTeacher() && resource.data.role == 'student')
        || isGuardianOf(userId);
      allow write: if false;
    }

    match /attendance/{attendanceId} {
      allow create, update: if isAdmin() || isTeacher();
      allow read: if isAdmin() || isTeacher()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /assessments/{assessmentId} {
      allow create, update: if isAdmin() || isTeacher();
      allow read: if isAdmin() || isTeacher()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /homework/{homeworkId} {
      allow create, update, delete: if isAdmin() || isTeacher();
      allow read: if request.auth != null;
    }

    match /fees/{feeId} {
      allow create, update, delete: if isAdmin();
      allow read: if isAdmin()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /schedule/{scheduleId} {
      allow create, update, delete: if isAdmin();
      allow read: if request.auth != null;
    }

    match /successStories/{storyId} {
      allow create, update, delete: if isAdmin();
      allow read: if resource.data.published == true || isAdmin();
    }
  }
}

(লক্ষ করুন: successStories-এ read rule চেক করে published true কিনা —
তাই Draft অবস্থায় থাকা গল্প admin ছাড়া আর কেউ দেখতে পারবে না, এমনকি
সরাসরি Firestore query করলেও না।)
