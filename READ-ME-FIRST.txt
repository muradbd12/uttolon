সবচেয়ে গুরুত্বপূর্ণ অংশ — Teacher/Student/Guardian account
তৈরির বাগ (এবার নিশ্চিত)
=========================================================
আপনার দেওয়া error message থেকে আসল কারণ পাওয়া গেছে:
firebase-admin-এর একটা ভেতরের অংশ (jose নামের একটা লাইব্রেরি) নতুন
ভার্সনে গিয়ে এমন ফরম্যাটে চলে গেছে যেটা Vercel-এর সার্ভারে ঠিকভাবে
লোড হয় না (ERR_REQUIRE_ESM error)।

সমাধান: package.json-এ এই লাইব্রেরিটাকে পুরনো, কাজ-করা একটা ভার্সনে
আটকে (pin করে) দিয়েছি। আমি নিজে সরাসরি টেস্ট করে নিশ্চিত হয়েছি এই
মডিউলটা এখন ঠিকভাবে লোড হচ্ছে এবং Firebase-এর নিজস্ব লজিক পর্যন্ত
পৌঁছাচ্ছে (আগে যেটা module লোড হওয়ার সময়ই ভেঙে পড়তো)।

**এই ফিক্সের জন্য package.json এবং package-lock.json — দুটোই বদলেছে,
দুটোই বসাতে হবে। package-lock.json বসানো এবার বিশেষভাবে জরুরি,
কারণ এটাই নিশ্চিত করে ঠিক কোন ভার্সনটা ব্যবহার হবে।**

ফাইল বসানোর পর টার্মিনালে একবার লিখুন (নতুন ভার্সন ডাউনলোড করতে):
  npm install

তারপর push করুন।

এর পাশাপাশি — বৃত্তি (Scholarship) আবেদন সিস্টেম
=========================================================
=== Security Rules-এ আরেকটা collection যোগ করুন ===
আগেরটার সাথে শুধু "scholarshipApplications" ব্লক নতুন:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isTeacher() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'teacher';
    }

    function isGuardianOf(studentUid) {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.linkedStudentUid == studentUid;
    }

    match /admissions/{applicationId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /notices/{noticeId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /users/{userId} {
      allow read: if (request.auth != null && request.auth.uid == userId)
        || isAdmin()
        || (isTeacher() && resource.data.role == 'student')
        || isGuardianOf(userId);
      allow write: if false;
    }

    match /attendance/{attendanceId} {
      allow create, update: if isAdmin() || isTeacher();
      allow read: if isAdmin() || isTeacher()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /assessments/{assessmentId} {
      allow create, update: if isAdmin() || isTeacher();
      allow read: if isAdmin() || isTeacher()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /homework/{homeworkId} {
      allow create, update, delete: if isAdmin() || isTeacher();
      allow read: if request.auth != null;
    }

    match /fees/{feeId} {
      allow create, update, delete: if isAdmin();
      allow read: if isAdmin()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /schedule/{scheduleId} {
      allow create, update, delete: if isAdmin();
      allow read: if request.auth != null;
    }

    match /successStories/{storyId} {
      allow create, update, delete: if isAdmin();
      allow read: if resource.data.published == true || isAdmin();
    }

    match /scholarshipApplications/{applicationId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }
  }
}

=========================================================
সবশেষে টেস্ট
=========================================================
1. npm install চালান, push করুন, Vercel deploy শেষ হওয়া পর্যন্ত
   অপেক্ষা করুন
2. /admin/users -এ গিয়ে Teacher অ্যাকাউন্ট তৈরির চেষ্টা করুন — এবার
   কাজ করা উচিত
3. /scholarship পেজে একটা টেস্ট আবেদন জমা দিয়ে /admin/scholarships
   থেকে অনুমোদন/প্রত্যাখ্যান করে দেখুন

যদি Teacher অ্যাকাউন্ট তৈরি এবারও ব্যর্থ হয়, তাহলে যা এখন দেখাবে
(সাদা বাক্সের লেখা) সেটা দিন — কিন্তু এবার আমি বেশ নিশ্চিত এটাই ছিল
আসল কারণ।
