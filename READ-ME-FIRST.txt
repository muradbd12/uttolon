এই আপডেটে অনেকগুলো ছোট-বড় real-data উন্নতি একসাথে —

১) Homework (হোমওয়ার্ক) সিস্টেম — Teacher একটা ক্লাসের জন্য হোমওয়ার্ক
   দিতে পারবেন, সেই ক্লাসের সব শিক্ষার্থী তাদের ড্যাশবোর্ডে দেখতে পাবে
২) Teacher Dashboard-এর "Recovery প্রয়োজন" তালিকা এখন real (Assessment
   থেকে যাদের Recovery Active চিহ্নিত করা হয়েছে তাদের দেখায়)
৩) Assessment ফর্মে এখন ঐচ্ছিক "মন্তব্য" ফিল্ড যোগ হয়েছে — Guardian
   Dashboard-এ এই মন্তব্য এখন real দেখাবে (demoTeacherComments বাদ)
৪) Student/Guardian/Teacher — তিন ড্যাশবোর্ডেই "নোটিশ" অংশ এখন সাইটের
   real নোটিশ বোর্ড থেকে আসে (আগে থেকেই থাকা Admin-প্রকাশিত নোটিশ)

=========================================================
ধাপ ১ — Security Rules-এ আরেকটা collection যোগ করুন
=========================================================
আগেরটার সাথে শুধু "homework" ব্লক নতুন — সব মুছে এটা দিয়ে Publish করুন:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isTeacher() {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'teacher';
    }

    function isGuardianOf(studentUid) {
      return request.auth != null &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.linkedStudentUid == studentUid;
    }

    match /admissions/{applicationId} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /notices/{noticeId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /users/{userId} {
      allow read: if (request.auth != null && request.auth.uid == userId)
        || isAdmin()
        || (isTeacher() && resource.data.role == 'student')
        || isGuardianOf(userId);
      allow write: if false;
    }

    match /attendance/{attendanceId} {
      allow create, update: if isAdmin() || isTeacher();
      allow read: if isAdmin() || isTeacher()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /assessments/{assessmentId} {
      allow create, update: if isAdmin() || isTeacher();
      allow read: if isAdmin() || isTeacher()
        || (request.auth != null && resource.data.studentUid == request.auth.uid)
        || isGuardianOf(resource.data.studentUid);
    }

    match /homework/{homeworkId} {
      allow create, update, delete: if isAdmin() || isTeacher();
      allow read: if request.auth != null;
    }
  }
}

টেস্ট করবেন যেভাবে
=========================================================
1. Teacher দিয়ে লগইন করে /teacher/homework -এ একটা ক্লাসের জন্য
   হোমওয়ার্ক দিন
2. সেই ক্লাসের একজন Student দিয়ে লগইন করে dashboard-এ হোমওয়ার্কটা
   দেখুন
3. /teacher/assessments-এ কোনো একটা মূল্যায়নে "Recovery Active" টিক
   দিয়ে একটা মন্তব্য লিখে সংরক্ষণ করুন
4. Teacher dashboard-এ ফিরে দেখুন "Recovery প্রয়োজন" তালিকায় সেই
   শিক্ষার্থী দেখাচ্ছে কিনা
5. সেই শিক্ষার্থীর Guardian দিয়ে লগইন করে "শিক্ষকের মন্তব্য"-এ
   মন্তব্যটা দেখা যাচ্ছে কিনা দেখুন
6. Admin দিয়ে একটা নোটিশ প্রকাশ করলে Student/Guardian/Teacher —
   তিন ড্যাশবোর্ডেই সেটা দেখা উচিত
